
-- 0. Krijimi i databazës dhe përdorimi i saj
CREATE DATABASE AutoPartsStore1;
GO

USE AutoPartsStore1;
GO

-- 1. Manufacturer
CREATE TABLE Manufacturer (
    ManufacturerID INT PRIMARY KEY IDENTITY(1,1),
    Name NVARCHAR(100) NOT NULL,
    Country NVARCHAR(100)
);

-- 2. CarModel
CREATE TABLE CarModel (
    CarModelID INT PRIMARY KEY IDENTITY(1,1),
    ManufacturerID INT,
    ModelName NVARCHAR(100) NOT NULL,
    YearStart INT,
    YearEnd INT,
    FOREIGN KEY (ManufacturerID) REFERENCES Manufacturer(ManufacturerID)
);

-- 3. Category
CREATE TABLE Category (
    CategoryID INT PRIMARY KEY IDENTITY(1,1),
    Name NVARCHAR(100) NOT NULL
);

-- 4. Part
CREATE TABLE Part (
    PartID INT PRIMARY KEY IDENTITY(1,1),
    CategoryID INT,
    Name NVARCHAR(200) NOT NULL,
    Description NVARCHAR(MAX),
    PartNumber NVARCHAR(100) UNIQUE,
    Manufacturer NVARCHAR(100),
    CompatibleFromYear INT,
    CompatibleToYear INT,
    FOREIGN KEY (CategoryID) REFERENCES Category(CategoryID)
);

-- 5. PartCarModel (Many-to-Many)
CREATE TABLE PartCarModel (
    PartID INT,
    CarModelID INT,
    PRIMARY KEY (PartID, CarModelID),
    FOREIGN KEY (PartID) REFERENCES Part(PartID),
    FOREIGN KEY (CarModelID) REFERENCES CarModel(CarModelID)
);

-- 6. Stock
CREATE TABLE Stock (
    StockID INT PRIMARY KEY IDENTITY(1,1),
    PartID INT,
    Quantity INT NOT NULL,
    Price DECIMAL(10,2) NOT NULL,
    Discount DECIMAL(5,2) DEFAULT 0,
    LastUpdated DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (PartID) REFERENCES Part(PartID)
);

-- 7. AppUser
CREATE TABLE AppUser (
    UserID INT PRIMARY KEY IDENTITY(1,1),
    FirstName NVARCHAR(100),
    LastName NVARCHAR(100),
    Email NVARCHAR(150) UNIQUE,
    Phone NVARCHAR(20)
);

-- 8. Role
CREATE TABLE Role (
    RoleID INT PRIMARY KEY IDENTITY(1,1),
    RoleName NVARCHAR(50) NOT NULL UNIQUE
);

-- 9. UserRole
CREATE TABLE UserRole (
    UserID INT,
    RoleID INT,
    PRIMARY KEY (UserID, RoleID),
    FOREIGN KEY (UserID) REFERENCES AppUser(UserID),
    FOREIGN KEY (RoleID) REFERENCES Role(RoleID)
);

-- 10. Address
CREATE TABLE Address (
    AddressID INT PRIMARY KEY IDENTITY(1,1),
    UserID INT,
    AddressLine NVARCHAR(255),
    City NVARCHAR(100),
    Country NVARCHAR(100),
    CreatedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (UserID) REFERENCES AppUser(UserID)
);

-- 11. Cart
CREATE TABLE Cart (
    CartID INT PRIMARY KEY IDENTITY(1,1),
    UserID INT,
    CreatedAt DATETIME DEFAULT GETDATE(),
    Status NVARCHAR(50) DEFAULT 'Active', -- 'Active', 'Ordered', 'Abandoned'
    FOREIGN KEY (UserID) REFERENCES AppUser(UserID)
);

-- 12. CartItem
CREATE TABLE CartItem (
    CartItemID INT PRIMARY KEY IDENTITY(1,1),
    CartID INT,
    PartID INT,
    Quantity INT NOT NULL CHECK (Quantity > 0),
    FOREIGN KEY (CartID) REFERENCES Cart(CartID),
    FOREIGN KEY (PartID) REFERENCES Part(PartID)
);

-- 13. Order
CREATE TABLE [Order] (
    OrderID INT PRIMARY KEY IDENTITY(1,1),
    CustomerID INT,
    OrderDate DATETIME DEFAULT GETDATE(),
    Status NVARCHAR(50) DEFAULT 'Pending',
    TotalAmount DECIMAL(10,2),
    FOREIGN KEY (CustomerID) REFERENCES AppUser(UserID)
);

-- 14. OrderDetail
CREATE TABLE OrderDetail (
    OrderDetailID INT PRIMARY KEY IDENTITY(1,1),
    OrderID INT,
    PartID INT,
    Quantity INT NOT NULL,
    UnitPrice DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (OrderID) REFERENCES [Order](OrderID),
    FOREIGN KEY (PartID) REFERENCES Part(PartID)
);

-- 15. OrderHistory
CREATE TABLE OrderHistory (
    OrderHistoryID INT PRIMARY KEY IDENTITY(1,1),
    OrderID INT,
    Status NVARCHAR(50),
    ChangedAt DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (OrderID) REFERENCES [Order](OrderID)
);
